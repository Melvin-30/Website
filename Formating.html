<!DOCTYPE html> <html> <head> <title>SOP 3 - Count Vowels</title> </head> <body> <h2>Count Vowels in a String</h2> <form name="f1"> <input type="text" name="String"> <input type="button" value="Check No of vowels" onclick="checkvowels()"> </form> <script> function checkvowels() { var str = f1.String.value, count = 0; str = str.toLowerCase(); for (var i = 0; i < str.length; i++) { ch=str[i]; if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u') { count++; } } alert("Number of vowels: " + count); } </script> </body> </html>
